print(f"RNA: {seq.transcribe()}")  # AUGCGUACGUAG

# 6. Translation with table (mitochondrial)
print(f"Mitochondrial: {seq.translate(table=2)}")

# 7. Count nucleotides

    print(f"Name: {record.name}")

    print(f"Length: {len(record.seq)}")

from Bio.PDB import PDBParser

# Create parser
parser = PDBParser(QUIET=True)  # QUIET suppresses warnings

# Parse structure
structure = parser.get_structure('protein', 'pdb_files/1XYZ.pdb')

# Basic structure hierarchy
print(f"Structure: {structure}")
print(f"Number of models: {len(structure)}")

# Navigate hierarchy: Structure -> Model -> Chain -> Residue -> Atom
for model in structure:
    print(f"Model: {model.id}")
    for chain in model:
        print(f"  Chain: {chain.id}")
        print(f"  Number of residues: {len(chain)}")
        for residue in chain:
            print(f"    Residue: {residue.resname} {residue.id[1]}")
            for atom in residue:
                print(f"      Atom: {atom.name} at {atom.coord}")

# Practical: Count everything
def count_structure_elements(structure):
    stats = {
        'models': 0,
        'chains': 0,
        'residues': 0,
        'atoms': 0
    }
    for model in structure:
        stats['models'] += 1
        for chain in model:
            stats['chains'] += 1
            for residue in chain:
                stats['residues'] += 1
                stats['atoms'] += len(residue)
    return stats

stats = count_structure_elements(structure)
print(stats)
